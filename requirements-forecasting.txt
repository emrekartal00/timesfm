# =============================================================================
# Everything the forecasting scripts need.
#
# From an IPython console:
#     %pip install -r requirements-forecasting.txt
#
# From a terminal:
#     python -m pip install -r requirements-forecasting.txt
#
# These are MINIMUM versions, not exact ones, deliberately. Pinning exact
# versions is what causes:
#
#     preparing metadata (pyproject.toml) did not run successfully
#     failed to activate VS environment: no vswhere.exe
#
# That error means pip could not find a ready-built package for your Python and
# tried to compile it from source, which on Windows needs Visual Studio C++
# tooling that you almost certainly do not have. Minimum versions let pip pick a
# ready-built one for whatever Python you are running, so nothing is compiled.
#
# If you hit that error anyway, your Python is probably too new for one of these
# packages. Check with:  import sys; print(sys.version)
# Python 3.11 or 3.12 is the safest choice today.
# =============================================================================

# --- The model itself --------------------------------------------------------
# On an NVIDIA machine, install torch from https://pytorch.org first to get the
# CUDA build, then run this file. The CPU build is slower but perfectly correct.
torch>=2.3

# Downloading and reading the TimesFM checkpoint. Not needed when loading
# weights from a local folder, but harmless.
huggingface_hub>=0.28
safetensors>=0.5

# --- Reading your spreadsheet ------------------------------------------------
pandas>=2.2
openpyxl>=3.1          # this is what lets pandas open .xlsx files
numpy>=1.26            # numpy 2 is fine; the whole stack was tested on 2.5.3

# --- The comparison model ----------------------------------------------------
lightgbm>=4.4          # the gradient-boosted model, --model gbm
scikit-learn>=1.6      # lightgbm imports this even when unused

# --- Seasonality report and plots (both optional) ----------------------------
# Without statsmodels you lose the autocorrelation and seasonal-strength lines.
# Without matplotlib you lose --plot. Everything else still runs.
statsmodels>=0.14.4
matplotlib>=3.9

# --- Public holidays ---------------------------------------------------------
# Supplies the holiday covariates, including Turkey's religious holidays, which
# move about eleven days earlier each year and so cannot be hard-coded.
# Without this the holiday features switch off silently rather than crashing --
# and switching them off cost 42% accuracy in testing, so do install it.
holidays>=0.60
